// ============================================================================
// Helper function to unpack a Little-Endian 4-byte float safely.
// ============================================================================
static float unpack_float_le(const uint8_t *src) {
    uint32_t temp = ((uint32_t)src[0]) |
                    ((uint32_t)src[1] << 8) |
                    ((uint32_t)src[2] << 16) |
                    ((uint32_t)src[3] << 24);
    float val;
    memcpy(&val, &temp, sizeof(float));
    return val;
}


// ============================================================================
// Helper function to copy a float into a byte array in Little-Endian format.
// ============================================================================
static void pack_float_le(uint8_t *dest, float val) {
    uint32_t temp;
    memcpy(&temp, &val, sizeof(float));
    dest[0] = (uint8_t)(temp & 0xFF);
    dest[1] = (uint8_t)((temp >> 8) & 0xFF);
    dest[2] = (uint8_t)((temp >> 16) & 0xFF);
    dest[3] = (uint8_t)((temp >> 24) & 0xFF);
}


// ============================================================================
// CRC-16/GSM Implementation
// Equivalent to CRC-16/XMODEM (Init 0x0000) with a final XOR of 0xFFFF.
// Polynomial: 0x1021, RefIn: False, RefOut: False
// ============================================================================
uint16_t crc16_gsm(const uint8_t *data, size_t length) {
    uint16_t crc = 0x0000; // XMODEM Init
    
    for (size_t i = 0; i < length; i++) {
        crc ^= (uint16_t)data[i] << 8;
        for (int j = 0; j < 8; j++) {
            if (crc & 0x8000) {
                crc = (crc << 1) ^ 0x1021;
            } else {
                crc <<= 1;
            }
        }
    }
    
    return crc ^ 0xFFFF; // GSM Final XOR
}


// ---------------------------------------------------------------------------
// Volume dB to Payload Conversion
// ---------------------------------------------------------------------------
void db_to_volume_payload(float db, uint8_t *out_payload) {
    double linear_gain = pow(10.0, db / 20.0);
    uint32_t raw_value = (uint32_t)llround(linear_gain * 8388608.0);
    
    if (raw_value > 0xFFFFFF) {
        raw_value = 0xFFFFFF;
    }
    
    out_payload[0] = 0x1F;
    out_payload[1] = (raw_value >> 16) & 0xFF;
    out_payload[2] = (raw_value >> 8)  & 0xFF;
    out_payload[3] = raw_value & 0xFF;
}


// ============================================================================
// Unescaping Helper (Includes Address Byte)
// ============================================================================
size_t unescape_frame(const uint16_t *frame, uint16_t frame_len, uint8_t *out_unescaped) {
    if (frame_len == 0) {
        return 0;
    }
    size_t unescaped_len = 0;
    
    // Write Address byte (strip 9th bit)
    out_unescaped[unescaped_len++] = frame[0] & 0xFF;
    
    // Process rest of the frame up to (but not including) the 7E terminator
    for (size_t i = 1; i < frame_len - 1; i++) {
        if (unescaped_len >= MAX_FRAME_SIZE) {
            break;
        }
        uint8_t val = frame[i] & 0x00FF; 
        
        if (val == 0x7D) {
            i++; 
            if (i >= frame_len - 1) {
                break;
            }
            uint8_t next_val = frame[i] & 0x00FF;
            if (next_val == 0x5E) {
                out_unescaped[unescaped_len++] = 0x7E;
            } else if (next_val == 0x5D) {
                out_unescaped[unescaped_len++] = 0x7D;
            }
        } else {
            out_unescaped[unescaped_len++] = val;
        }
    }
    return unescaped_len;
}


// --- Command Handlers ---
void print_payload_hex(const uint8_t *payload, size_t len) {
    printf("Payload (%u bytes): ", len);
    for (size_t i = 0; i < len; i++) {
        printf("%02X ", payload[i]);
    }
    printf("\r\n");
}

__weak void on_set_volume(const uint8_t *payload, size_t len) {
    if (len >= 4 && payload[0] == 0x1F) {
        // Extract the 24-bit Big-Endian fixed-point gain
        uint32_t raw_value = ((uint32_t)payload[1] << 16) | 
                             ((uint32_t)payload[2] << 8) | 
                             (uint32_t)payload[3];
        
        if (raw_value == 0) {
            printf("  -> [Handler] Set Volume: -inf dB (Muted)\r\n");
        } else {
            // Convert to linear gain, then to dB
            double linear_gain = (double)raw_value / 8388608.0;
            double db = 20.0 * log10(linear_gain);
            int16_t db_int, db_dec;
            db_int = round(db * 10);
            db_int = db_int / 10;
            db_dec = (round(db * 10) - (db_int * 10)) * -1;
            printf("  -> [Handler] Set Volume: %d.%d dB\r\n", db_int, db_dec);
        }
    } else {
        printf("  -> [Handler] Set Volume Payload received (%u bytes)\r\n", len);
    }
}

__weak void unknown_message(const uint16_t *frame, uint16_t frame_len) {
    static char print_buf[MAX_PRINT_BUFFER];
    size_t offset = 0;
    for (size_t i = 0; i < frame_len; i++) {
        if (frame[i] >= 0x0100) {
            offset += snprintf(&print_buf[offset], MAX_PRINT_BUFFER - offset, "0x%03X ", frame[i]);
        } else {
            offset += snprintf(&print_buf[offset], MAX_PRINT_BUFFER - offset, "0x%02X ", frame[i] & 0xFF);
        }
        if (offset >= MAX_PRINT_BUFFER - 10) { 
            strcat(print_buf, "...(truncated)"); 
            break; 
        }
    }
    printf(" [Handler] UNKNOWN MESSAGE: %s \r\n", print_buf);
}


// ============================================================================
// Handles incoming unescaped PEQ parameters, extracts biquad coefficients,
// reverse-calculates original UI values, and prints them.
// ============================================================================
void on_peq(uint8_t session_id, const uint8_t *peq_data, size_t len) {
    if (!peq_data || len < 21) {
        printf("  -> [Handler] Error: Invalid PEQ payload length\r\n");
        return;
    }

    uint8_t index = peq_data[0];
    
    // Unpack fields from the already-unescaped clean payload buffer
    float b0 = unpack_float_le(&peq_data[1]);
    float b1 = unpack_float_le(&peq_data[5]);
    float b2 = unpack_float_le(&peq_data[9]);
    float raw_a1 = unpack_float_le(&peq_data[13]); 
    float raw_a2 = unpack_float_le(&peq_data[17]); 

    // --- DYNAMIC FILTER NAME ASSIGNMENT ---
    char filter_name[32];
    if (index == 0) {
        strcpy(filter_name, "Low Shelf 1");
    } else if (index == 1) {
        strcpy(filter_name, "Low Shelf 2");
    } else if (index == 2) {
        strcpy(filter_name, "High Shelf 1");
    } else if (index == 3) {
        strcpy(filter_name, "High Shelf 2");
    } else {
        // Dynamic numbering for notches (Index 4 maps to Notch 1, Index 5 to Notch 2, etc.)
        snprintf(filter_name, sizeof(filter_name), "Notch %d", index - 3);
    }

    // --- HANDLE BYPASSED / FLAT SLOTS ---
    if (fabsf(b0 - 1.0f) < 1e-4f && fabsf(b1) < 1e-4f && fabsf(b2) < 1e-4f) {
        float def_freq = 1000.0f;
        float def_q = 1.0f;
        
        if (index == 0)       { def_freq = 60.0f;   def_q = 1.0f; }
        else if (index == 1)  { def_freq = 100.0f;  def_q = 1.0f; }
        else if (index == 2)  { def_freq = 150.0f;  def_q = 2.0f; }
        else if (index == 3)  { def_freq = 604.5f;  def_q = 1.0f; }
        else if (index == 11) { def_freq = 728.9f;  def_q = 2.0f; }
        else if (index >= 4 && index <= 10) { def_freq = 150.0f; def_q = 2.0f; }
        
        int32_t df_scaled = (int32_t)roundf(def_freq * 10.0f);
        int32_t dq_scaled = (int32_t)roundf(def_q * 10.0f);
        
        printf("  -> 0x%02X, %s, %ld.%ld Hz, 0.0 dB, %ld.%ld (Bypassed)\r\n", 
               session_id, filter_name, 
               (long)(df_scaled / 10), (long)(df_scaled % 10),
               (long)(dq_scaled / 10), (long)(dq_scaled % 10));
        return;
    }

    float frequency = 0.0f;
    float gain = 0.0f;
    float q = 1.0f;

    // Normalization mapping verified by working Notch filters
    float a1_norm = -raw_a1;
    float a2_norm = -raw_a2;

    // --- REVERSE-ENGINEER COEFFS BY FILTER TYPE ---
    if (index <= 3) {
        // --- SHELF EQ PROCESSING (Indices 0, 1, 2, 3) ---
        float H = 1.0f;
        
        if (index == 2 || index == 3) {
            // High Shelf (1 & 2): Evaluate response at Nyquist frequency (z = -1)
            float denom = 1.0f + a1_norm - a2_norm;
            if (fabsf(denom) > 1e-6f) {
                H = (b0 - b1 + b2) / denom;
            }
        } else {
            // Low Shelf (1 & 2): Evaluate response at DC (z = 1)
            float denom = 1.0f + a1_norm + a2_norm;
            if (fabsf(denom) > 1e-6f) {
                H = (b0 + b1 + b2) / denom;
            }
        }
        
        if (H < 0.0f) H = -H;
        gain = 20.0f * log10f(H);

        float A = sqrtf(H);
        if (A < 1e-4f) A = 1e-4f;
        
        if (index == 2 || index == 3) {
            // High Shelf Frequency Extraction
            float denom_a0 = a1_norm + (b1 / A);
            if (fabsf(denom_a0) > 1e-5f) {
                float a0 = (4.0f * (A - 1.0f)) / denom_a0;
                float cos_w0 = ((b1 / A) - a1_norm) * a0 / (4.0f * (A + 1.0f));
                
                if (cos_w0 < 0.0f) cos_w0 = -cos_w0; 
                if (cos_w0 > 1.0f)  cos_w0 = 1.0f;
                if (cos_w0 < -1.0f) cos_w0 = -1.0f;
                frequency = (acosf(cos_w0) * SAMPLE_RATE) / (2.0f * (float)M_PI);
            }
        } else {
            // Low Shelf Frequency Extraction
            float denom_a0 = a1_norm - (b1 / A);
            if (fabsf(denom_a0) > 1e-5f) {
                float a0 = (4.0f * (A - 1.0f)) / denom_a0;
                float cos_w0 = ((b1 / A) + a1_norm) * a0 / (-4.0f * (A + 1.0f));
                
                if (cos_w0 < 0.0f) cos_w0 = -cos_w0; 
                if (cos_w0 > 1.0f)  cos_w0 = 1.0f;
                if (cos_w0 < -1.0f) cos_w0 = -1.0f;
                frequency = (acosf(cos_w0) * SAMPLE_RATE) / (2.0f * (float)M_PI);
            }
        }
        q = 1.0f; 
    } else {
        // --- NOTCH / PEAKING EQ PROCESSING (Indices >= 4) ---
        float num = b0 - b2;
        float den = 1.0f - a2_norm;
        if (fabsf(den) > 1e-6f) {
            float A_sq = num / den;
            if (A_sq > 0.0f) {
                float A = sqrtf(A_sq);
                gain = 40.0f * log10f(A);
                
                float a0 = 2.0f / (1.0f + a2_norm);
                float cos_w0 = (-a1_norm * a0) / 2.0f;
                if (cos_w0 > 1.0f)  cos_w0 = 1.0f;
                if (cos_w0 < -1.0f) cos_w0 = -1.0f;
                
                float w0 = acosf(cos_w0);
                frequency = (w0 * SAMPLE_RATE) / (2.0f * (float)M_PI);
                
                float denom_q = 2.0f * A * (1.0f - a2_norm);
                if (fabsf(denom_q) > 1e-6f) {
                    q = (sinf(w0) * (1.0f + a2_norm)) / denom_q;
                }
            }
        }
    }

    // --- FIXED-POINT MATH CONVERSION FOR PRINT HACK ---
    // it doesn't require printf('%f') which requires a much larger library
    int32_t f_scaled = (int32_t)roundf(frequency * 10.0f);
    int32_t f_int = f_scaled / 10;
    int32_t f_dec = f_scaled % 10;

    int32_t g_scaled = (int32_t)roundf(gain * 10.0f);
    int32_t g_int = g_scaled / 10;
    int32_t g_dec = g_scaled % 10;
    if (g_dec < 0) g_dec = -g_dec;

    int32_t q_scaled = (int32_t)roundf(q * 10.0f);
    int32_t q_int = q_scaled / 10;
    int32_t q_dec = q_scaled % 10;

    // Output formatted stream using clean, integer-only format tokens
    printf("  -> 0x%02X, %s, %ld.%ld Hz, ", session_id, filter_name, (long)f_int, (long)f_dec);

    if (g_scaled < 0 && g_int == 0) {
        printf("-0.%ld dB, ", (long)g_dec);
    } else {
        printf("%ld.%ld dB, ", (long)g_int, (long)g_dec);
    }

    printf("%ld.%ld\r\n", (long)q_int, (long)q_dec);
}


// ============================================================================
// Helper to un-escape 16-bit array into 8-bit array (excluding address and 7E)
// ============================================================================
size_t unescape_frame_to_payload(const uint16_t *frame, uint16_t frame_len, uint8_t *out_unescaped) {
    size_t unescaped_len = 0;
    for (size_t i = 1; i < frame_len - 1; i++) {
        if (unescaped_len >= MAX_FRAME_SIZE) {
            break;
        }
        uint8_t val = frame[i] & 0x00FF; 
        if (val == 0x7D) {
            i++; 
            if (i >= frame_len - 1) {
                break;
            }
            uint8_t next_val = frame[i] & 0x00FF;
            if (next_val == 0x5E) {
                out_unescaped[unescaped_len++] = 0x7E;
            } else if (next_val == 0x5D) {
                out_unescaped[unescaped_len++] = 0x7D;
            }
        } else {
            out_unescaped[unescaped_len++] = val;
        }
    }
    return unescaped_len;
}


// ============================================================================
// Frame Validator (Validates CRC over Address + Payload)
// ============================================================================
bool validate_frame_16bit(const uint16_t *frame, uint16_t frame_len, 
                          uint8_t *out_payload, size_t *out_payload_len) {
    
    if (frame_len < 4 || frame[frame_len - 1] != 0x007E) {
        return false;
    }

    uint8_t unescaped[MAX_FRAME_SIZE];
    size_t unescaped_len = unescape_frame(frame, frame_len, unescaped);

    // Must have at least Addr (1) + CRC (2) = 3 bytes
    if (unescaped_len < 3) {
        return false;
    }

    // Extract received CRC (Last 2 bytes, Big-Endian: High byte, then Low byte)
    uint16_t received_crc = (unescaped[unescaped_len - 2] << 8) | unescaped[unescaped_len - 1];
    
    // Calculate CRC over Address + Payload (everything except the 2 CRC bytes)
    uint16_t calculated_crc = crc16_gsm(unescaped, unescaped_len - 2);

    if (out_payload && out_payload_len) {
        *out_payload_len = unescaped_len - 3; // Subtract Addr (1) and CRC (2)
        memcpy(out_payload, &unescaped[1], *out_payload_len); // Skip Address byte
    }

    return (received_crc == calculated_crc);
}


// ============================================================================
// Frame Builder (Calculates CRC over Address + Payload)
// ============================================================================
int build_frame_16bit(uint16_t address_9bit, const uint8_t *payload, size_t payload_len, 
                      uint16_t *out_frame, size_t out_max_len) {
    
    if (!payload || !out_frame || out_max_len < payload_len + 4) {
        return -1;
    }

    // 1. Create a temporary array of Address + payload
    uint8_t temp_buf[MAX_FRAME_SIZE];
    if (payload_len + 1 > MAX_FRAME_SIZE) {
        return -1;
    }
    
    temp_buf[0] = address_9bit & 0xFF; // Address lower 8 bits
    memcpy(&temp_buf[1], payload, payload_len);

    // 2. Calculate CRC over Address + Payload
    uint16_t crc = crc16_gsm(temp_buf, payload_len + 1);
    uint8_t crc_hi = (crc >> 8) & 0xFF;
    uint8_t crc_lo = crc & 0xFF;

    size_t idx = 0;

    // 3. Write Address byte (Set 9th bit)
    out_frame[idx++] = address_9bit; 

    // Helper macro to write data to the 16-bit buffer with HDLC escaping
    #define WRITE_BYTE(b) do { \
        if (idx >= out_max_len - 1) return -1; \
        if ((b) == 0x7E) { \
            out_frame[idx++] = 0x007D; \
            out_frame[idx++] = 0x005E; \
        } else if ((b) == 0x7D) { \
            out_frame[idx++] = 0x007D; \
            out_frame[idx++] = 0x005D; \
        } else { \
            out_frame[idx++] = (uint16_t)(b); \
        } \
    } while(0)

    // 4. Write Payload
    for (size_t i = 0; i < payload_len; i++) {
        WRITE_BYTE(payload[i]);
    }

    // 5. Write CRC (Big-Endian: High byte, then Low byte)
    WRITE_BYTE(crc_hi);
    WRITE_BYTE(crc_lo);

    // 6. Write End Flag (0x007E)
    if (idx >= out_max_len) {
        return -1;
    }
    out_frame[idx++] = 0x007E;

    #undef WRITE_BYTE
    return (int)idx;
}


// =====================================================================================
// Converts standard PEQ filter definition and sending it to the Genelec speakers
// =====================================================================================
// @param index        The filter index (0x00 to 0x13).
// @param type         The type of filter (Low Shelf, High Shelf, or Peaking).
// @param frequency    Center/cutoff frequency in Hz.
// @param gain_db      Gain in decibels.
// @param q            Quality factor (used for Peaking/Notch filters).
// @param out_payload  Pointer to a buffer of at least 24 bytes.
// @return             0 on success, -1 on invalid parameters.
// ============================================================================
int generate_peq_payload(uint8_t index, FilterType type, float frequency, float gain_db, float q, uint8_t *out_payload) {
    if (!out_payload || index > 0x13) {
        return -1;
    }
    if (frequency <= 0.0f || frequency >= (SAMPLE_RATE / 2.0f)) {
        return -1;
    }

    // Fixed payload frame header setup
    out_payload[0] = 0x10;       // Command byte
    out_payload[1] = 0x0E;       // Sub-command (Set PEQ)
    out_payload[2] = index;      // Filter Index

    // --- UNIVERSAL BYPASSED / FLAT SCENARIO ---
    // If the gain is 0.0 dB, any filter type collapses to a flat unity gain line.
    // The hardware optimizes this into a clean identity matrix passing b0=1.0.
    if (fabsf(gain_db) < 1e-4f) {
        pack_float_le(&out_payload[3],  1.0f); // b0 = 1.0
        pack_float_le(&out_payload[7],  0.0f); // b1 = 0.0
        pack_float_le(&out_payload[11], 0.0f); // b2 = 0.0
        pack_float_le(&out_payload[15], 0.0f); // a1 = 0.0
        pack_float_le(&out_payload[19], 0.0f); // a2 = 0.0
        out_payload[23] = 0x00;                // Fixed trailing Type/Flag byte
        return 0;
    }

    // High-precision double intermediate math variables to minimize rounding drift
    double w0 = 2.0 * M_PI * (double)frequency / (double)SAMPLE_RATE;
    double cos_w0 = cos(w0);
    double sin_w0 = sin(w0);
    
    double A = pow(10.0, (double)gain_db / 40.0);
    double alpha = 0.0;

    double b0 = 0.0, b1 = 0.0, b2 = 0.0;
    double a0 = 1.0, a1 = 0.0, a2 = 0.0;

    switch (type) {
        case FILTER_PEAKING:
            if (q <= 0.0f) {
                return -1;
            }
            alpha = sin_w0 / (2.0 * (double)q);
            b0 = 1.0 + alpha * A;
            b1 = -2.0 * cos_w0;
            b2 = 1.0 - alpha * A;
            a0 = 1.0 + alpha / A;
            a1 = -2.0 * cos_w0;
            a2 = 1.0 - alpha / A;
            break;

        case FILTER_LOW_SHELF:
            // Hardcoded shelf slope S = 1.0 simplifies alpha term to:
            alpha = (sin_w0 / 2.0) * sqrt(2.0);

            b0 = A * ((A + 1.0) - (A - 1.0) * cos_w0 + 2.0 * sqrt(A) * alpha);
            b1 = 2.0 * A * ((A - 1.0) - (A + 1.0) * cos_w0);
            b2 = A * ((A + 1.0) - (A - 1.0) * cos_w0 - 2.0 * sqrt(A) * alpha);
            a0 = (A + 1.0) + (A - 1.0) * cos_w0 + 2.0 * sqrt(A) * alpha;
            a1 = -2.0 * ((A - 1.0) + (A + 1.0) * cos_w0);
            a2 = (A + 1.0) + (A - 1.0) * cos_w0 - 2.0 * sqrt(A) * alpha;
            break;

        case FILTER_HIGH_SHELF:
            // Hardcoded shelf slope S = 1.0 simplifies alpha term to:
            alpha = (sin_w0 / 2.0) * sqrt(2.0);

            b0 = A * ((A + 1.0) + (A - 1.0) * cos_w0 + 2.0 * sqrt(A) * alpha);
            b1 = -2.0 * A * ((A - 1.0) + (A + 1.0) * cos_w0);
            b2 = A * ((A + 1.0) + (A - 1.0) * cos_w0 - 2.0 * sqrt(A) * alpha);
            a0 = (A + 1.0) - (A - 1.0) * cos_w0 + 2.0 * sqrt(A) * alpha;
            a1 = 2.0 * ((A - 1.0) - (A + 1.0) * cos_w0);
            a2 = (A + 1.0) - (A - 1.0) * cos_w0 - 2.0 * sqrt(A) * alpha;
            break;

        default:
            return -1;
    }

    // Division guard to prevent NaN payloads
    if (fabs(a0) < 1e-9) {
        return -1;
    }

    // Normalization & Sign Inversion matching the DSP decoder core
    pack_float_le(&out_payload[3],  (float)(b0 / a0));
    pack_float_le(&out_payload[7],  (float)(b1 / a0));
    pack_float_le(&out_payload[11], (float)(b2 / a0));
    pack_float_le(&out_payload[15], (float)(-(a1 / a0))); // Sign Inverted
    pack_float_le(&out_payload[19], (float)(-(a2 / a0))); // Sign Inverted

    // Corrected placement index for the trailing frame type byte
    out_payload[23] = 0x00; 

    return 0;
}


// ============================================================================
// Troubleshooting Printer (Updated for robustness)
// ============================================================================
void print_message_for_troubleshooting(const uint16_t *frame, uint16_t bytes_read) {
    // Check if we at least have an address byte and a terminator
    if (bytes_read < 4 || frame[bytes_read - 1] != 0x007E) {
        printf("no terminator or incomplete frame\r\n");
        return;
    }

    uint8_t unescaped[MAX_FRAME_SIZE];
    size_t unescaped_len = unescape_frame_to_payload(frame, bytes_read, unescaped);

    if (unescaped_len < 2) {
        printf("%03X (too short)\r\n", frame[0]);
        return;
    }

    size_t payload_len = unescaped_len - 2;
    uint16_t calculated_crc = crc16_gsm(unescaped, payload_len);

    // print raw length
    printf(" %d ", bytes_read);

    // 1. Print 9-bit command in hex
    printf("%03X ", frame[0]);

    // 2. Print payload in hex
    for (size_t i = 0; i < payload_len; i++) {
        printf("%02X ", unescaped[i]);
    }

    // 3. Print CRC in the message, followed by calculated CRC in ()
    printf("CRC %02X %02X (%04X) - ", unescaped[unescaped_len - 2], unescaped[unescaped_len - 1], calculated_crc);
}


int send_flash_commit(uint8_t sess_id, const FlashCommitSettings_t *settings) {
    if (!settings) return -1;
    if (settings->max_level_db > 0.0f || settings->max_level_db < -20.0f) return -2;
    if (settings->startup_level_db > 0.0f || settings->startup_level_db < -60.0f) return -3;
    if (settings->iss_sensitivity > 2) return -4;
    if (settings->input_select > 2) return -5;

    uint16_t unicast_addr = 0x0100 | sess_id;
    uint8_t payload[8];

    // 1. Max Level Restriction (0x10 01 00 [3-byte])
    // Math TBD, using approximation: raw = 10^(db/20) * 0x7FFFFF
    payload[0] = 0x10; payload[1] = 0x01; payload[2] = 0x00;
    float max_lin = powf(10.0f, settings->max_level_db / 20.0f);
    uint32_t max_raw = (uint32_t)llroundf(max_lin * 8388607.0f);
    payload[3] = (max_raw >> 16) & 0xFF;
    payload[4] = (max_raw >> 8) & 0xFF;
    payload[5] = max_raw & 0xFF;
    send_frame(unicast_addr, payload, 6);

    // 2. Startup Level (0x10 01 05 [3-byte])
    payload[2] = 0x05;
    float start_lin = powf(10.0f, settings->startup_level_db / 20.0f);
    uint32_t start_raw = (uint32_t)llroundf(start_lin * 8388607.0f);
    payload[3] = (start_raw >> 16) & 0xFF;
    payload[4] = (start_raw >> 8) & 0xFF;
    payload[5] = start_raw & 0xFF;
    send_frame(unicast_addr, payload, 6);

    // 3. ISS Sleep Delay (0x3A 01 [4-byte BE])
    payload[0] = 0x3A; payload[1] = 0x01;
    payload[2] = (settings->iss_sleep_delay >> 24) & 0xFF;
    payload[3] = (settings->iss_sleep_delay >> 16) & 0xFF;
    payload[4] = (settings->iss_sleep_delay >> 8) & 0xFF;
    payload[5] = settings->iss_sleep_delay & 0xFF;
    send_frame(unicast_addr, payload, 6);

    // 4. ISS Sensitivity (0x3A 02 [4-byte])
    payload[1] = 0x02;
    uint32_t sens_val = 0xFFFFFFB5; // High
    if (settings->iss_sensitivity == 1) sens_val = 0xFFFFFFBC; // Medium
    if (settings->iss_sensitivity == 2) sens_val = 0xFFFFFFC3; // Low
    payload[2] = (sens_val >> 24) & 0xFF;
    payload[3] = (sens_val >> 16) & 0xFF;
    payload[4] = (sens_val >> 8) & 0xFF;
    payload[5] = sens_val & 0xFF;
    send_frame(unicast_addr, payload, 6);

    // 5. LED State (0x3A 06 [1-byte])
    payload[1] = 0x06;
    payload[2] = settings->led_on ? 0x00 : 0x01;
    send_frame(unicast_addr, payload, 3);

    // 6. Standalone Config Block (0x40 00 [Input] 02 [Counter])
    payload[0] = 0x40; payload[1] = 0x00;
    payload[2] = settings->input_select + 1; // 1=Ana, 2=Dig, 3=Auto
    payload[3] = 0x02;
    payload[4] = settings->config_counter;
    send_frame(unicast_addr, payload, 5);

    // 7. Commit to Flash (0x15 33 00)
    payload[0] = 0x15; payload[1] = 0x33; payload[2] = 0x00;
    send_frame(unicast_addr, payload, 3);

    return 0;
}


// ============================================================================
// PROTOCOL COMMAND SUMMARY
// ============================================================================
// Addressing:
//   0x01FF : Hub Broadcast (Used by PC App for global commands)
//   0x01F0 : Hub Multicast (Used by Hub standalone / Session assignment)
//   0x0102 - 0x01FE : Hub Unicast (Lower 8 bits = Speaker Session ID)
//   0x0101 : Speaker Response (All speaker replies use this address)
//
// Notes:
//   - CRC is CRC-16/GSM (Poly 0x1021, Init 0x0000, XorOut 0xFFFF).
//   - CRC is calculated over Address + Payload, appended in Big-Endian order.
//   - Speaker payloads always start with 0x09. A 0x06 wrapper/sequence byte
//     may follow the 0x09 header before the actual response data.
//   - All multi-byte values are Big-Endian unless otherwise specified.
//
// ============================================================================
// 1. HUB COMMANDS (Hub -> Speakers)
// ----------------------------------------------------------------------------
// CMD  | TARGET  | PAYLOAD FORMAT                  | DESCRIPTION
// -----|---------|---------------------------------|---------------------------
// 0x02 | 0x01F0  | 02 [ID1] [ID2] [ID3] [SessID]  | Assign Session ID
// 0x04 | 0x01FF  | 04 BC 84                        | Broadcast Config / Sync
// 0x05 | Unicast | 05 04 00 00 00 FF FF EA ...     | Unicast DSP Config
// 0x08 | Unicast | 08                              | Telemetry Poll
// 0x10 | Unicast | 10 [Sub] [Data...]              | DSP Parameter Update
//      |         |   Sub 0E: [Idx] [5x Float32 LE] |   Set PEQ Filter
//      |         |   Sub 02: 00 00 00 00           |   DSP Config 02
//      |         |   Sub 01: 00 7F FF FF ...       |   DSP Config 01
// 0x17 | Unicast | 17 01 [Data...]                 | Unicast Config
// 0x19 | Unicast | 19 01                           | Serial Number Request
// 0x1F | 0x01FF/ | 1F [B2] [B1] [B0]              | Set Volume
//      | 0x01F0  | (24-bit BE fixed-point gain)    | (1.0 gain = 0x800000)
// 0x2B | Unicast | 2B [03 / 04]                    | Set Mute (03=ON, 04=OFF)
// 0x39 | Unicast | 39                              | System Info Request
// 0x3A | 0x01FF/ | 3A 03 [02 / 00]                 | System Power (02=ON,00=OFF)
//      | 0x01F0  |                                 |
// 0x3B | Unicast | 3B 00 01                        | Unicast Config (Bass Mgmt)
// 0x3D | Unicast | 3D 00 00                        | Unicast Config (Input Sel)
// 0x40 | Unicast | 40 00 01 02 00                  | Unicast Config
// 0xFE | 0x01FF  | FE                              | Discovery Probe
//
// ============================================================================
// 2. SPEAKER RESPONSES (Speaker -> Hub, Address = 0x0101)
// ----------------------------------------------------------------------------------------
// RESP | PAYLOAD FORMAT                                                  | DESC
// -----|-----------------------------------------------------------------|----------------
// 0x09 | 09 [SessID]                                                     | Session ID ACK
// 0x09 | 09 [ID1] [ID2] [ID3]                                            | Hardware ID Resp
// 0x09 | 09 5D E7                                                        | Generic ACK
// 0x41 | 09 [06] 41 [Temp] 81 00 [VU1] 83 00 [VU2] 42 [VU3] 43 [VU4]     | Telemetry Resp
//      | 45 [VU5] 47 [State] 84 02 [Vol]                                 |
// 0x63 | 09 [06] 63 [ASCII String \0]                                    | System Info Resp
// ASCII| 09 [06] [16-byte ASCII Serial]                                  | Serial Num Resp
// ============================================================================
 
void parse_message(uint16_t *rx_data, uint16_t bytes_read) {
    if (bytes_read == 0 || rx_data[bytes_read - 1] != 0x007E) {
        printf("Parser Warning: Incomplete frame or no 7E terminator\r\n");
        if (bytes_read > 0) {
            print_message_for_troubleshooting(rx_data, bytes_read);
        }
        return; 
    }

    uint16_t frame_len = bytes_read;
    uint8_t payload[MAX_FRAME_SIZE];
    size_t payload_len = 0;

    printf("%" PRIu32 " ", bsp_millis());

    if (!validate_frame_16bit(rx_data, frame_len, payload, &payload_len)) {
        printf("Troubleshooting ");
        print_message_for_troubleshooting(rx_data, frame_len);
        crc_message(rx_data, frame_len);
        return;
    } 

    if (payload_len < 1) {
        unknown_message(rx_data, frame_len);
        return;
    }

    uint16_t addr = rx_data[0];
    uint8_t sess_id = addr & 0xFF; 

    // --- HUB BROADCAST (0x01FF) & HUB MULTICAST (0x01F0) ---
    if (addr == 0x01FF || addr == 0x01F0) {
        switch (payload[0]) {
            case 0xFE: 
                if (addr == 0x01FF) {
                    on_hub_probe(payload, payload_len); 
                } else {
                    unknown_message(rx_data, frame_len);
                }
                break;
            case 0x3A: 
                if (payload_len >= 3) {
                    on_system_power(payload[2] == 0x02);
                } else {
                    unknown_message(rx_data, frame_len);
                }
                break;
            case 0x1F: 
                on_set_volume(payload, payload_len); 
                break;
            case 0x02: 
                if (addr == 0x01F0 && payload_len >= 5) {
                    on_hub_assign_session(&payload[1], payload[4], payload, payload_len);
                } else {
                    unknown_message(rx_data, frame_len);
                }
                break;
            case 0x04:
                on_cmd_04(); // Broadcast config command
                break;
            default:   
                unknown_message(rx_data, frame_len); 
                break;
        }
    } 
    else if (addr >= 0x0102 && addr <= 0x01FE) {
        // --- HUB UNICAST ---
        switch (payload[0]) {
            case 0x08: on_telemetry_poll(sess_id); break;
            case 0x10: 
                if (payload_len >= 2) {
                    if (payload[1] == 0x0E) on_peq(sess_id, &payload[2], payload_len - 2); 
                    else if (payload[1] == 0x02) on_dsp_cmd_02(sess_id, &payload[2], payload_len - 2);
                    else if (payload[1] == 0x01) on_dsp_cmd_01(sess_id, &payload[2], payload_len - 2);
                    else unknown_message(rx_data, frame_len);
                } else {
                    unknown_message(rx_data, frame_len);
                }
                break;
            case 0x2B: 
                if (payload_len >= 2) on_mute(sess_id, payload[1] == 0x03);
                else unknown_message(rx_data, frame_len);
                break;
            case 0x19: 
                if (payload_len >= 2 && payload[1] == 0x01) on_serial_request(sess_id);
                else unknown_message(rx_data, frame_len);
                break;
            case 0x39: 
                // System Info Request. 
                // This is a 1-byte command. The changing bytes we saw earlier 
                // were just the CRC.
                on_info_request(sess_id);
                break;
            // New Unicast Config Commands
            case 0x05: on_cmd_05(sess_id, payload, payload_len); break;
            case 0x17: on_cmd_17(sess_id, payload, payload_len); break;
            case 0x3B: on_cmd_3B(sess_id, payload, payload_len); break;
            case 0x3D: on_cmd_3D(sess_id, payload, payload_len); break;
            case 0x40: on_cmd_40(sess_id, payload, payload_len); break;
            default:   unknown_message(rx_data, frame_len); break;
        }
    } 
    else if (addr == 0x0101) {
        // --- SPEAKER RESPONSE ---
        if (payload[0] == 0x09) {
            // --- NEW: handle Flash Write Complete (09 06) ---
            if (payload_len >= 2 && payload[1] == 0x06 && payload_len == 2) {
                on_flash_write_complete();
                return;
            }
            // Handle 1-byte Generic ACK (Payload is just 0x09)
            if (payload_len == 1) {
                on_generic_ack();
            } 
            else if (payload_len >= 2) {
                // Handle the 0x06 wrapper byte if present
                int offset = (payload[1] == 0x06) ? 1 : 0;
                
                if (payload_len <= 1 + offset) {
                    unknown_message(rx_data, frame_len);
                } else {
                    switch (payload[1 + offset]) {
                        case 0x41: on_telemetry_response(&payload[1 + offset], payload_len - 1 - offset); break;
                        case 0x63: on_info_response((const char*)&payload[1 + offset], payload_len - 1 - offset); break;
                        case 0x5D: 
                            if (payload_len >= 3 + offset && payload[2 + offset] == 0xE7) {
                                on_generic_ack();
                            } else {
                                unknown_message(rx_data, frame_len); 
                            }
                            break;
                        default:
                            // HW ID (Len 4) or Session ACK (Len 2) or Serial (ASCII)
                            if (payload_len - offset == 2) {
                                on_speaker_session_ack(payload[1 + offset], payload, payload_len);
                            } else if (payload_len - offset == 4) {
                                on_speaker_hw_id(&payload[1 + offset], payload, payload_len);
                            } else if (payload_len - offset > 4 && payload[1 + offset] >= 0x30 && payload[1 + offset] <= 0x7A) {
                                on_serial_response((const char*)&payload[1 + offset], payload_len - 1 - offset);
                            } else {
                                unknown_message(rx_data, frame_len);
                            }
                            break;
                    }
                }
            } else {
                unknown_message(rx_data, frame_len);
            }
        } else {
            unknown_message(rx_data, frame_len);
        }
    } 
    else {
        // --- UNKNOWN ADDRESS ---
        unknown_message(rx_data, frame_len);
    }
}


// ============================================================================
// dump_message function: print hex payloads on RS485 bus
// ============================================================================
void dump_message(uint16_t *rx_data, uint16_t bytes_read) {
    if (bytes_read == 0 || rx_data[bytes_read - 1] != 0x007E) {
        printf("Parser Warning: Incomplete frame or no 7E terminator\r\n");
        if (bytes_read > 0) {
            print_message_for_troubleshooting(rx_data, bytes_read);
        }
        return;
    }

    uint16_t frame_len = bytes_read;
    uint8_t payload[MAX_FRAME_SIZE];
    size_t payload_len = 0;

    printf("%" PRIu32 " ", bsp_millis());

    if (!validate_frame_16bit(rx_data, frame_len, payload, &payload_len)) {
        printf("Troubleshooting ");
        print_message_for_troubleshooting(rx_data, frame_len);
        crc_message(rx_data, frame_len);
        return;
    }

    if (payload_len < 1) {
        unknown_message(rx_data, frame_len);
        return;
    }

    // print raw length
    printf(" %2d ", bytes_read);

    static char print_buf[MAX_PRINT_BUFFER];
    size_t offset = 0;
    for (size_t i = 0; i < frame_len; i++) {
        if (rx_data[i] >= 0x0100) {
            offset += snprintf(&print_buf[offset], MAX_PRINT_BUFFER - offset,
                               "%03X ", rx_data[i]);
        } else {
            offset += snprintf(&print_buf[offset], MAX_PRINT_BUFFER - offset,
                               "%02X ", rx_data[i] & 0xFF);
        }
        if (offset >= MAX_PRINT_BUFFER - 10) {
            strcat(print_buf, "...(truncated)");
            break;
        }
    }
    printf("  %s\r\n", print_buf);
}
